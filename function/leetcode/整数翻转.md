字符串翻转以及数学解法

### 解题思路
1. 数学解法

   1. 例如x = 12345
      1. result = result*10 + x %10;x /= 10;
         1. `result`0*10 + 12345%10 = 5 `x`12345/10 = 1234
         2. `result`5*10 + 1234%10 = 54 `x`1234/10 = 123
         3. `result`54*10 + 123%10 = 543 `x`123/10 = 12
         4. `result`543*10 + 12%10 = 5432 `x`12/10 = 1
         5. `result`5432*10 + 1%10 = 54321 `x`1/10 = 0
         6. 结束
   2. 例如x = -12345
      1. result = result*10 + x %10;x /= 10;
         1. `result`0*10 - 12345%10 = -5 `x`-12345/10 = -1234
         2. `result`-5*10 - 1234%10 = -54 `x`-1234/10 = -123
         3. `result`-54*10 - 123%10 = -543 `x`-123/10 = -12
         4. `result`-543*10 - 12%10 = -5432 `x`-12/10 = -1
         5. `result`-5432*10 - 1%10 = -54321 `x`-1/10 = 0
         6. 结束

2. 字符串翻转解法

   1. 转换int为string

      1. 如果x为负数:转换负数为正数并将符号加入到result
      2. 否则直接转换

   2. 依次获取翻转的数字并且丢入result

   3. 转换为int类型

   4. 判断是否合法

      1. 合法:提交result
      2. 不合法:提交0

      




### 代码

```golang
func reverse(x int) int {
	var result int
	for x != 0 {
		result = result*10 + x%10
		x /= 10
	}
	if result < math.MinInt32 || result > math.MaxInt32 {
		return 0
	}
	return result
}
```

```golang
func reverse(x int) int {
	var str string
	var result []uint8
	if x < 0 {
		str = strconv.Itoa(x + (-x)*2)
		result = append(result, uint8('-'))
	} else {
		str = strconv.Itoa(x)
	}
	for i, _ := range str {
		result = append(result, str[len(str)-1-i])
	}
	resultInt, _ := strconv.Atoi(string(result))
    if resultInt <= math.MaxInt32 && resultInt >= math.MinInt32 {
        return resultInt
    }
    return 0
}
```
